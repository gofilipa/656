\subsection{BERT model}
\label{sec:orge9b148b}
\begin{itemize}
\item open source model from google
\begin{itemize}
\item October 2018
\item "encoder only" "transformer" architecture
\item trained primarily by masked token prediction
\item the goal of training is to come up with embeddings that are
context-aware
\end{itemize}
\item "encoder-only"
\begin{itemize}
\item it develops word embeddings, does not generate word predictions
(continuous words, like gpt)
\item masking words and making guesses
\item bi-directional, looks both ways
\end{itemize}
\item "transformer" 
\begin{itemize}
\item attention mechanism
\begin{itemize}
\item a way of assessing the strength of relationships between words
in context
\end{itemize}
\item \url{https://github.com/jessevig/bertviz}
\end{itemize}
\item lineage of models
\begin{itemize}
\item gpt1 -> bert -> gpt2 -> gpt3
\item \url{https://huggingface.co/learn/llm-course/chapter1/4}
\item the "berts"
\begin{itemize}
\item bert -> roberta -> bart -> deberta
\end{itemize}
\end{itemize}
\end{itemize}

# associate with "Masked language modeling (MLM)" randomly masks some
# tokens in the input and trains the model to predict the original
# tokens based on the surrounding context. This allows the model to
# learn bidirectional context (looking at words both before and after
# the masked word). These models use the bidirectional approach to
# understand context from both directions.

# They’re best suited for tasks that require deep understanding of
# text, such as classification, named entity recognition, and question
# answering.

# BERT is the first model to effectively implement deep
# bidirectionality to learn richer representations of the text by
# attending to words on both sides.

########################
### LOADING THE DATA ###
########################

# load the movie reviews
from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")

# examine the data, which has two fields:
#    text: the movie review text.
#    label: a value that is either 0 for a negative review or 1 for a positive review.
imdb["test"][0]

##########################
### PREPARING THE DATA ###
##########################

# now we will tokenize the data

# load distilbert tokenizer
from transformers import AutoTokenizer
tokenizer = AutoTokenizer.from_pretrained("distilbert/distilbert-base-uncased")

# Randomly sample 2,000 training examples
train_dataset = imdb["train"].shuffle(seed=42).select(range(2000))

# define our processing function for tokenization
def preprocess_function(examples):
    return tokenizer(examples["text"], truncation=True)

# now, tokenize
tokenized_train = train_dataset.map(
    preprocess_function,
    batched=True
)

# load our data collator, for padding the data to equal lengths
from transformers import DataCollatorWithPadding
data_collator = DataCollatorWithPadding(tokenizer=tokenizer)

# finally, create a map of the expected ids to their labels with id2label and label2id:
id2label = {0: "NEGATIVE", 1: "POSITIVE"}
label2id = {"NEGATIVE": 0, "POSITIVE": 1}

################
### TRAINING ###
################

# load the training libraries
from transformers import AutoModelForSequenceClassification, TrainingArguments, Trainer

# load the training model
model = AutoModelForSequenceClassification.from_pretrained(
    "distilbert/distilbert-base-uncased", num_labels=2, id2label=id2label, label2id=label2id
)

# define our training arguments
training_args = TrainingArguments(
    output_dir="my_awesome_model",
    learning_rate=2e-5,
    per_device_train_batch_size=16,
    per_device_eval_batch_size=16,
    num_train_epochs=2,
    weight_decay=0.01,
)

# define the trainer
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_train,
    processing_class=tokenizer,
    data_collator=data_collator,
)

# train the model!
trainer.train()

# save the model locally
trainer.save_model('./bert-finetuned')

#################
### INFERENCE ###
#################

# create a text for testing it out (whatever you want can go in the quotes)
text = "This was a masterpiece. Not completely faithful to the books, but enthralling from beginning to end. Might be my favorite of the three."

# load the pipeline for inference
from transformers import pipeline

# define our classifier with the newest model
classifier = pipeline("sentiment-analysis", model="./bert-finetuned")

# run the classifer
classifier(text)

# try out different prompts
text1 = "What a boring film. Waste of time."
classifier(text1)

texts = ['Loved it', 'It was meh', 'I have seen better', 'It was too far long', 'It stimulated me', 'Visuals', 'Dialogue', 'Character development']
for i in texts:
    print(classifier(i))
